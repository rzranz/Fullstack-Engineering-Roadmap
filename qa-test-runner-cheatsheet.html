2. Pemetaan Core Web Vitals & Web Performance (LCP, INP, CLS, TTFB, FCP, TBT)

Dalam QA Performance Engineering modern, pengujian terbagi menjadi dua layer yang saling melengkapi: Backend/Protocol Load Testing (k6 API — Modul 3) dan Frontend User-Centric Metrics (Core Web Vitals di Browser).

Metrik Kepanjangan & Definisi Ambang Batas Google Posisi di Proyek Ini & Cara Mengukurnya
TTFB Time to First Byte — Waktu sejak permintaan HTTP dikirim hingga byte pertama respons diterima dari server/CDN. Good ≤ 800 ms
Warn 800–1800 ms
Poor > 1800 ms		 Sudah terukur di Modul 3! Pada k6, metrik ini bernama http_req_waiting (hasil kita: avg 229.05 ms, p95 415.14 ms — masuk kategori Good).
LCP Largest Contentful Paint — Waktu hingga elemen konten terbesar di viewport (hero image, heading utama, tabel/banner besar) selesai ter-render. Good ≤ 2.5 s
Warn 2.5–4.0 s
Poor > 4.0 s		 Layer Browser Frontend. Dipengaruhi TTFB server, render-blocking CSS/JS, dan ukuran aset. Diukur via Lighthouse, Chrome DevTools Performance, atau k6/browser.
INP Interaction to Next Paint (pengganti resmi FID sejak Maret 2024) — Latensi terlama (p98) dari klik/ketik/tap user hingga browser menggambar frame visual berikutnya. Good ≤ 200 ms
Warn 200–500 ms
Poor > 500 ms		 Layer Browser Frontend. Dipengaruhi Long Tasks (>50 ms) di main thread JS, state update React yang berat, atau event handler sinkron. (Contoh relevan di OpenMRS: gejala hilangnya ketikan awal di input Vitals adalah problem responsivitas input di sisi UI!)
CLS Cumulative Layout Shift — Total pergeseran posisi elemen visual secara tiba-tiba saat halaman sedang dimuat/digunakan. Good ≤ 0.10
Warn 0.10–0.25
Poor > 0.25		 Layer Browser Frontend. Dipicu gambar/iframe tanpa dimensi tetap, webfont late-swap, atau banner/toast dinamis yang mendorong konten ke bawah.
FCP & TBT First Contentful Paint (teks/gambar pertama muncul) & Total Blocking Time (total durasi main thread terkunci >50 ms sebelum interaktif). FCP ≤ 1.8 s
TBT ≤ 200 ms		 Proksi Lab di Lighthouse: Karena lab audit tidak selalu menekan semua tombol secara nyata, TBT digunakan Lighthouse sebagai indikator kuat potensi buruknya INP di dunia nyata.

Kapan Pakai k6 Protocol Test (Modul 3)?

  • Menguji daya tahan backend/database/CDN pada 500–10.000+ concurrent user.
  • Mengukur RPS, HTTP error rate, connection pooling, TLS, dan TTFB (http_req_waiting).
  • Cepat dan ringan (1 mesin laptop bisa menjalankan 1.000 VUs sekaligus).

Kapan Pakai Web Vitals / Lighthouse / k6/browser?

  • Mengukur pengalaman visual & interaksi user di browser (LCP, INP, CLS).
  • Menemukan bundle JS terlalu besar, main thread blocking, atau layout shift.
  • Umumnya dijalankan pada 1–10 browser instances (karena tiap Chromium memakan RAM & CPU besar), bukan 1.000 VUs.

/**
 * RestAssured API automation tests against ReqRes (https://reqres.in/).
 */
public class ReqResApiTest {

    private static final String BASE_URI = "https://reqres.in";
    private static final String API_KEY = System.getProperty("reqres.api.key", "free_user_3MMnHzsnIN28CqZwivUmSkD0fi9");

    @BeforeClass
    public void setup() {
        RestAssured.baseURI = BASE_URI;
    }

    @Test(priority = 1, description = "Scenario 1: GET /api/users/2 returns HTTP 200 and data.id equals '2'")
    public void testGetSingleUserById() {
        Response response = given()
                .header("x-api-key", API_KEY)
                .accept(ContentType.JSON)
                .when()
                .get("/api/users/2")
                .then()
                .log().all()
                .extract().response();

        Reporter.log("GET /api/users/2 Status: " + response.statusCode(), true);
        Reporter.log("GET /api/users/2 Body: " + response.asString(), true);

        Assert.assertEquals(response.statusCode(), 200, "Expected HTTP 200 OK");
        Assert.assertEquals(response.jsonPath().getString("data.id"), "2",
                "Expected data.id to equal '2'");
    }

    @Test(priority = 2, description = "Scenario 2: POST /api/login without password returns HTTP 400 and error 'Missing password'")
    public void testUnsuccessfulLoginMissingPassword() {
        Response response = given()
                .header("x-api-key", API_KEY)
                .contentType(ContentType.JSON)
                .accept(ContentType.JSON)
                .body(Collections.singletonMap("email", "peter@klaven"))

    }

Java Page Object — OpenMrsLoginPage.java (Selenium 4 + OWASP CSRFGuard Wait)
src/test/java/com/gits/qa/pages/OpenMrsLoginPage.java 
package com.gits.qa.pages;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

import java.time.Duration;
import java.util.List;

public class OpenMrsLoginPage {

    public static final String BASE_URL = "https://o3.openmrs.org/openmrs/";

    private static final By USERNAME_INPUT = By.id("username");
    private static final By PASSWORD_INPUT = By.id("password");
    private static final By LOGIN_SUBMIT_BUTTON = By.cssSelector("input[type='submit'][value='Log In'], form button[type='submit']");
    private static final By SESSION_LOCATION_LIST = By.id("sessionLocation");
    private static final By ERROR_MESSAGE_CONTAINER = By.id("openmrs_error");

    private static final By LOGOUT_LINK = By.linkText("Log out");
    private static final By LOGGED_IN_USER_BAR = By.id("userLoggedInAs");
    private static final By NAV_LIST = By.cssSelector("ul.navList");

    private final WebDriver driver;
    private final WebDriverWait wait;

    public OpenMrsLoginPage(WebDriver driver) {
        this.driver = driver;
        this.wait = new WebDriverWait(driver, Duration.ofSeconds(20));
    }

    public void open() {
        driver.get(BASE_URL);
        wait.until(ExpectedConditions.visibilityOfElementLocated(USERNAME_INPUT));
        wait.until(ExpectedConditions.presenceOfElementLocated(By.name("OWASP-CSRFTOKEN")));
    }

    public void enterUsername(String username) {
        WebElement el = wait.until(ExpectedConditions.visibilityOfElementLocated(USERNAME_INPUT));
        el.clear();
        el.sendKeys(username);
    }

    public void enterPassword(String password) {
        WebElement el = wait.until(ExpectedConditions.visibilityOfElementLocated(PASSWORD_INPUT));
        el.clear();
        el.sendKeys(password);
    }

    public boolean selectLocationIfRequired(String locationName) {
        if (locationName == null || locationName.isBlank()) {
            return false;
        }
        List<WebElement> locationLists = driver.findElements(SESSION_LOCATION_LIST);
        if (!locationLists.isEmpty() && locationLists.get(0).isDisplayed()) {
            WebElement option = locationLists.get(0).findElement(
                    By.xpath(".//li[normalize-space()='" + locationName + "']"));
            option.click();
            return true;
        }
        return false;
    }

    public void clickLogin() {
        WebElement submitBtn = wait.until(ExpectedConditions.elementToBeClickable(LOGIN_SUBMIT_BUTTON));
        submitBtn.click();
    }

    public boolean login(String username, String password, String locationName) {
        enterUsername(username);
        enterPassword(password);
        boolean locationSelected = selectLocationIfRequired(locationName);
        clickLogin();
        return locationSelected;
    }

    public String getErrorMessageText() {
        WebElement err = wait.until(ExpectedConditions.visibilityOfElementLocated(ERROR_MESSAGE_CONTAINER));
        return err.getText().trim();
    }

    public boolean isDashboardDisplayed() {
        wait.until(ExpectedConditions.and(
                ExpectedConditions.visibilityOfElementLocated(LOGOUT_LINK),
                ExpectedConditions.visibilityOfElementLocated(NAV_LIST)
        ));
        List<WebElement> userBar = driver.findElements(LOGGED_IN_USER_BAR);
        boolean hasUserBar = !userBar.isEmpty() && userBar.get(0).isDisplayed();
        return driver.findElement(LOGOUT_LINK).isDisplayed()
                && driver.findElement(NAV_LIST).isDisplayed()
                && (hasUserBar || driver.getPageSource().contains("Welcome to OpenMRS"));
    }

    public String getLoggedInUserText() {
        List<WebElement> userBar = driver.findElements(LOGGED_IN_USER_BAR);
        return userBar.isEmpty() ? "" : userBar.get(0).getText().trim();
    }

    public void logoutIfLoggedIn() {
        List<WebElement> logoutLinks = driver.findElements(LOGOUT_LINK);
        if (!logoutLinks.isEmpty() && logoutLinks.get(0).isDisplayed()) {
            logoutLinks.get(0).click();
            wait.until(ExpectedConditions.visibilityOfElementLocated(USERNAME_INPUT));
        }
    }
}

import http from 'k6/http';
import { check, sleep, group } from 'k6';
import { Trend, Rate } from 'k6/metrics';
import exec from 'k6/execution';

const getReqDuration = new Trend('get_req_duration', true);
const postReqDuration = new Trend('post_req_duration', true);
const getReqFailed = new Rate('get_req_failed');
const postReqFailed = new Rate('post_req_failed');

const phase500Duration = new Trend('phase_500vu_duration', true);
const phase1000Duration = new Trend('phase_1000vu_duration', true);
const phase500Failed = new Rate('phase_500vu_failed');
const phase1000Failed = new Rate('phase_1000vu_failed');

const BASE_URL = 'https://jsonplaceholder.typicode.com';

export const options = {
  stages: [
    { duration: '1m', target: 500 },
    { duration: '1m', target: 500 },
    { duration: '1m', target: 1000 },
    { duration: '1m', target: 1000 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    http_req_duration: ['p(95)<2000'],
    http_req_failed: ['rate<0.05'],
  },
};

function recordPhaseMetrics(durationMs, isFailed) {
  const elapsedMs = Date.now() - exec.scenario.startTime;
  if (elapsedMs <= 120000) {
    phase500Duration.add(durationMs);
    phase500Failed.add(isFailed);
  } else if (elapsedMs <= 240000) {
    phase1000Duration.add(durationMs);
    phase1000Failed.add(isFailed);
  }
}

export default function () {
  group('GET /posts/1', function () {
    const getRes = http.get(`${BASE_URL}/posts/1`, {
      tags: { endpoint: 'GET_posts_1' },
    });

    const getOk = check(getRes, {
      'GET /posts/1 status is 200': (r) => r.status === 200,
      'GET /posts/1 body id === 1': (r) => {
        if (r.status !== 200) return false;
        try {
          return r.json('id') === 1;
        } catch (e) {
          return false;
        }
      },
    });

    getReqDuration.add(getRes.timings.duration);
    getReqFailed.add(!getOk);
    recordPhaseMetrics(getRes.timings.duration, !getOk);
  });

  sleep(1);

  group('POST /posts', function () {
    const payloadObj = {
      title: `GITS QA Load Test VU-${__VU}`,
      body: `Performance iteration ${__ITER}`,
      userId: (__VU % 10) + 1,
    };
    const params = {
      headers: { 'Content-Type': 'application/json; charset=UTF-8' },
      tags: { endpoint: 'POST_posts' },
    };

    const postRes = http.post(`${BASE_URL}/posts`, JSON.stringify(payloadObj), params);

    const postOk = check(postRes, {
      'POST /posts status is 201': (r) => r.status === 201,
      'POST /posts payload matches': (r) => {
        if (r.status !== 201) return false;
        try {
          const data = r.json();
          return (
            data.title === payloadObj.title &&
            data.body === payloadObj.body &&
            Number(data.userId) === payloadObj.userId &&
            data.id !== undefined
          );
        } catch (e) {
          return false;
        }
      },
    });

    postReqDuration.add(postRes.timings.duration);
    postReqFailed.add(!postOk);
    recordPhaseMetrics(postRes.timings.duration, !postOk);
  });

  sleep(1);
}
